function cellVal(c){
  if(!c) return '';
  if(c.t==='n') return c.v; if(c.t==='d') return c.v; if(c.t==='b') return String(c.v);
  if(c.t==='e'||c.t==='z') return '';
  return c.v!==undefined&&c.v!==null?c.v:(c.w||'');
}
function extractRaw(sheet){
  let rows=[];
  const dense=Array.isArray(sheet)?sheet:(Array.isArray(sheet&&sheet['!data'])?sheet['!data']:null);
  if(dense){ for(let r=0;r<dense.length;r++){ const rc=dense[r]; if(!rc){ rows.push(null); continue; } const a=new Array(rc.length); for(let c=0;c<rc.length;c++) a[c]=cellVal(rc[c]); rows.push(a); } }
  else rows=XLSX.utils.sheet_to_json(sheet,{header:1,raw:true,defval:'',blankrows:true});
  let h=-1; for(let r=0;r<rows.length;r++){ if(rows[r]&&rows[r].some(v=>String(v).trim()!=='')){ h=r; break; } }
  if(h<0) return {headers:[],data:[],srcRows:[]};
  const headers=rows[h].map(v=>norm(String(v)));
  const data=[], srcRows=[];
  for(let r=h+1;r<rows.length;r++){ const a=rows[r]; if(!a||!a.some(v=>String(v).trim()!=='')) continue; data.push(a); srcRows.push(r+1); }
  return {headers,data,srcRows};
}
function sheetFromWb(f,name){
  if(f.rawBytes){ const wb=XLSX.read(f.rawBytes,{type:'array',sheets:[name],...AUDIT_XLSX_OPTS}); return wb.Sheets[name]; }
  return f.wb.Sheets[name];
}
function loadSheetInto(f,name){
  const raw=extractRaw(sheetFromWb(f,name));
  if(!raw.headers.length) throw new Error('برگهٔ «'+name+'» خالی است.');
  f.sheet=name; f.headers=raw.headers; f.data=raw.data; f.srcRows=raw.srcRows;
  f.mapping=guessMapping(f.headers); f.keyFields=['taf','taf1','taf2'].filter(k=>f.mapping[k]>=0);
  const prev=SR.files.find(x=>x!==f&&x.headers&&x.headers.join('|')===f.headers.join('|'));
  if(prev){ f.mapping={...prev.mapping}; f.keyFields=prev.keyFields.slice(); }
  guessYear(f);
}
function guessMapping(headers){
  // هر ستون فقط به یک فیلد داده می‌شود: ابتدا قوی‌ترین تطابق‌ها
  const g=bestGuesses(headers), map={}, cand=[];
  Object.keys(FIELD_EXPECTED).forEach(k=>{ map[k]=-1; const i=EXPECTED.indexOf(FIELD_EXPECTED[k]); if(i>=0&&g[i]&&g[i].idx>=0) cand.push({k,idx:g[i].idx,score:g[i].score,ord:Object.keys(FIELD_EXPECTED).indexOf(k)}); });
  cand.sort((a,b)=>b.score-a.score||a.ord-b.ord);
  const used=new Set();
  cand.forEach(c=>{ if(!used.has(c.idx)){ map[c.k]=c.idx; used.add(c.idx); } });
  if(map.taf<0&&map.taf1>=0&&map.taf2<0){ map.taf=map.taf1; map.taf1=-1; if(map.tafT<0){ map.tafT=map.taf1T; map.taf1T=-1; } }
  return map;
}
function guessYear(f){
  if(f.mapping.date<0){ f.start=null; f.end=null; return; }
  let lo=null, hi=null;
  for(const r of f.data){ const p=SRE.parseDate(r[f.mapping.date]); if(p.ok){ if(lo===null||p.day<lo) lo=p.day; if(hi===null||p.day>hi) hi=p.day; } }
  f.minDay=lo; f.maxDay=hi;
  if(lo===null){ f.start=null; f.end=null; return; }
  const oneYear=SRE.addMonths(lo,12)-1;
  f.start=lo; f.end=hi<=oneYear?oneYear:hi;
  f.label=fa(SRE.dayToJalali(f.end).jy);
}
async function addFiles(list){
  const host=$('srS1');
  for(const file of list){
    const f={id:'f'+(++SR.seq),name:file.name,status:'reading',hasOpening:true,openingDocs:new Set(),closingDocs:new Set()};
    SR.files.push(f); renderFiles();
    try{
      const wb=await auditReadWorkbookFile(file);
      f.sheetNames=wb.SheetNames.slice();
      if(wb._rawBytes){ f.rawBytes=wb._rawBytes; f.wb=null; }
      else f.wb=wb;
      let best=f.sheetNames[0];
      if(f.sheetNames.length>1){ let bs=-1; f.sheetNames.forEach(n=>{ const s=sheetScore(wb,n).score; if(s>bs){ bs=s; best=n; } }); }
      loadSheetInto(f,best);
      f.status='ok';
    }catch(err){ f.status='error'; f.error=auditWorkbookMessage?auditWorkbookMessage(err):String(err&&err.message||err); }
    if(!SR.auditId&&f.status==='ok') SR.auditId=f.id;
    SR.mapped=false; invalidate('model');
    if(host&&CUR_STEP==='s1') renderFiles();
  }
}

function canonRows(f){

}

function buildModel(){
  const files=SR.files.filter(f=>f.status==='ok').map(f=>({id:f.id,label:f.label||f.name,name:f.name,start:f.start,end:f.end,rows:f.canon,srcRows:f.srcRows,

  SR.model=SRE.build({files,auditFileId:SR.auditId});
  const sug=SRE.suggestRoles(SR.model);
  sug.forEach((v,k)=>{ if(!SR.roles.has(k)) SR.roles.set(k,v); });
  SR.codingSug=SRE.suggestCoding(SR.model);
  SR.codingOk=new Set([...SR.codingOk].filter(id=>SR.codingSug.some(s=>s.id===id)));
  SR.result=null;
}

function runAnalysis(){
  const accounts=new Set([...SR.roles.entries()].filter(([,r])=>r.selected).map(([k])=>k));
  const links=SR.linkSug.filter(l=>SR.linkOk.has(l.id)).map(l=>({origEid:l.origEid,laterEid:l.laterEid}));
  const aliases=SR.codingSug.filter(s=>SR.codingOk.has(s.id)).map(s=>({fromKey:s.fromKey,toKey:s.toKey,toFileId:s.toFileId}));
  SR.result=SRE.analyze(SR.model,{roles:SR.roles,tagOverrides:SR.tagOverrides,confirmedTags:new Set(SR.tagOverrides.keys()),links,aliases,accounts,
    thresholds:SR.settings.thresholds.slice(),cutoffMethod:SR.settings.cutoffMethod,inclusive:SR.settings.inclusive});
  SR.view.limit=200; SR.view.key=null; SR.view.T=0;
}

function printReport(){
  const src=$('srReport'); if(!src) return;
  if(typeof window.__activePrintRestore==='function') window.__activePrintRestore();
  const ph=$('printHost'); let done=false; const old=document.title;
  const rule=document.createElement('style'); rule.textContent='@page{size:A4 landscape;margin:8mm}';
  const restore=()=>{ if(done) return; done=true; if(window.__activePrintRestore===restore) window.__activePrintRestore=null; document.title=old; document.body.classList.remove('printing'); ph.innerHTML=''; rule.remove(); };
  ph.innerHTML=`<div class="sr-print" dir="rtl">${src.innerHTML}</div>`;
